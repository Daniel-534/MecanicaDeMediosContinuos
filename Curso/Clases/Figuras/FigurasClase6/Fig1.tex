    \centering
    \begin{tikzpicture}[>=Latex,font=\small,line cap=round,line join=round]
        \coordinate (O) at (0.65,0.55);
        \coordinate (S) at (4.25,3.55);
        \coordinate (P) at (10.15,4.75);

        \draw[very thick,fill=blue!8]
            plot[smooth cycle,tension=0.75] coordinates {
                (1.55,1.15) (1.85,3.60) (3.45,4.65)
                (5.35,4.15) (5.85,2.20) (4.70,0.90)
                (2.65,0.65)};
        \node[blue!65!black] at (3.45,4.15)
            {$\rho(\vec r^{\,\prime})$};

        \foreach \x/\y in {
            2.05/1.45,2.20/2.65,2.70/3.55,2.85/1.05,
            3.25/2.25,3.55/3.25,3.85/1.35,4.25/3.55,
            4.55/2.35,5.05/3.35,5.20/1.65}
        {
            \shade[ball color=blue!60] (\x,\y) circle (0.075);
        }

        \fill[black] (O) circle (0.07);
        \node[anchor=north east] at (O) {$O$};
        \fill[red!75!black] (S) circle (0.11);
        \fill[black] (P) circle (0.09);
        \node[anchor=south west] at ($(P)+(0.08,0.05)$)
            {punto de campo};

        \draw[->,ultra thick,blue!65!black]
            (O)--(S)
            node[pos=0.52,above left] {$\vec r^{\,\prime}$};
        \draw[->,ultra thick,black!75]
            (O)--(P)
            node[pos=0.66,below] {$\vec r$};
        \draw[->,ultra thick,red!75!black]
            (S)--(P)
            node[midway,above,sloped]
            {$\vec r-\vec r^{\,\prime}$};

        \draw[->,ultra thick,green!45!black]
            ($(P)+(0,-0.28)$)--++(-1.45,-0.30)
            node[midway,below] {$d\vec g$};

        \node[anchor=north,align=center] (dm) at (3.10,0.50)
            {$dm=\rho(\vec r^{\,\prime})\,d^3\vec r^{\,\prime}$};
        \draw[gray!70,thin] (dm.north east)--(S);

        \node[align=center,text=black!70] at (8.40,1.30)
            {El campo total se obtiene al sumar\\
             las contribuciones de todos los $dm$.};
    \end{tikzpicture}
    \caption{El vector $\vec r^{\,\prime}$ localiza el elemento de masa,
    $\vec r$ localiza el punto de campo y su diferencia
    $\vec r-\vec r^{\,\prime}$ une la
    fuente con el punto donde se calcula la contribución $d\vec g$.}
