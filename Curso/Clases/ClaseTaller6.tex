\chapter{Clase Taller 6}

\section{Problemas propuestos}

En este taller se emplea la convención de suma de Einstein.
\subsection*{Problema 2}

Partiendo de
\[
d\vec r=\sum_i h_i\hat e_i\,dq_i,
\]
demuestre que
\[
\text{(a)}\qquad
\frac{\partial\hat e_i}{\partial q_j}
=\frac{\hat e_j}{h_i}\frac{\partial h_j}{\partial q_i},
\qquad i\ne j,
\]
y que
\[
\text{(b)}\qquad
\frac{\partial\hat e_i}{\partial q_i}
=-\sum_{k\ne i}\frac{\hat e_k}{h_k}\frac{\partial h_i}{\partial q_k}.
\]

\textbf{Solución.} Como
\[
\frac{\partial\vec r}{\partial q_i}=h_i\hat e_i,
\]
la igualdad de las derivadas mixtas implica
\[
\frac{\partial}{\partial q_j}(h_i\hat e_i)
=\frac{\partial}{\partial q_i}(h_j\hat e_j).
\]
Para $i\ne j$, al proyectar esta igualdad sobre $\hat e_j$ y usar la ortonormalidad de la base, se obtiene
\[
h_i\hat e_j\cdot\frac{\partial\hat e_i}{\partial q_j}
=\frac{\partial h_j}{\partial q_i}.
\]
Las demás proyecciones se anulan; por tanto,
\[
\boxed{
\frac{\partial\hat e_i}{\partial q_j}
=\frac{\hat e_j}{h_i}\frac{\partial h_j}{\partial q_i}},
\qquad i\ne j.
\]
Además, $\hat e_i\cdot\hat e_i=1$ implica que
$\partial\hat e_i/\partial q_i$ no tiene componente en la dirección $\hat e_i$. Para $k\ne i$,
\[
\hat e_k\cdot\frac{\partial\hat e_i}{\partial q_i}
=-\hat e_i\cdot\frac{\partial\hat e_k}{\partial q_i}
=-\frac{1}{h_k}\frac{\partial h_i}{\partial q_k},
\]
donde se utilizó el resultado anterior. Así,
\[
\boxed{
\frac{\partial\hat e_i}{\partial q_i}
=-\sum_{k\ne i}\frac{\hat e_k}{h_k}
\frac{\partial h_i}{\partial q_k}}.
\]

\subsection*{Problema 3}

Escriba la regla de transformación entre los vectores de la base cartesiana
$(\hat\epsilon_1,\hat\epsilon_2,\hat\epsilon_3)$ y los vectores de una base arbitraria
$(\hat e_1,\hat e_2,\hat e_3)$.

\textbf{Solución.} Si $\mathbb A=\{a_{ij}\}$ es la matriz de cambio de base, entonces
\[
\boxed{\hat e_i=a_{ij}\hat\epsilon_j},
\qquad
a_{ij}=\hat e_i\cdot\hat\epsilon_j.
\]
La transformación inversa es
\[
\hat\epsilon_i=(\mathbb A^{-1})_{ij}\hat e_j.
\]
Si ambas bases son ortonormales, $\mathbb A^{-1}=\mathbb A^T$ y, por tanto,
\[
\boxed{\hat\epsilon_i=a_{ji}\hat e_j},
\qquad
a_{ik}a_{jk}=a_{ki}a_{kj}=\delta_{ij}.
\]

\subsection*{Problema 4}

Dados
\[
\mathbb S=
\begin{pmatrix}
1&0&2\\
0&1&2\\
3&0&3
\end{pmatrix},
\qquad
\vec a=
\begin{pmatrix}
1\\2\\3
\end{pmatrix},
\]
calcule:
\[
\text{(a) }S_{ii},
\qquad
\text{(b) }S_{ij}S_{ij},
\qquad
\text{(c) }S_{ij}S_{ji},
\]
\[
\text{(d) }a_ma_m,
\qquad
\text{(e) }S_{mn}a_ma_n.
\]
Escriba cada una de estas expresiones en notación matricial.

\textbf{Solución.}

La contracción de los índices de un tensor corresponde a su traza:
\[
\text{(a)}\qquad
S_{ii}=\mathrm{Tr}\,\mathbb S=\boxed{5}.
\]

Para $S_{ij}S_{ij}$ se introduce el tensor transpuesto, usando
$(\mathbb S^T)_{ji}=S_{ij}$:
\[
\begin{aligned}
\text{(b)}\qquad
S_{ij}S_{ij}
&=(\mathbb S^T)_{ji}S_{ij}\\
&=(\mathbb S^T\mathbb S)_{jj}\\
&=\mathrm{Tr}(\mathbb S^T\mathbb S)
=\boxed{28}.
\end{aligned}
\]

En la contracción cruzada aparece directamente el producto $\mathbb S^2$:
\[
\begin{aligned}
\text{(c)}\qquad
S_{ij}S_{ji}
&=(\mathbb S^2)_{ii}\\
&=\mathrm{Tr}(\mathbb S^2)
=\boxed{23}.
\end{aligned}
\]

Para el vector, la contracción se escribe como un producto fila--columna:
\[
\begin{aligned}
\text{(d)}\qquad
a_ma_m
&=(\vec a^{\,T})_m a_m\\
&=\vec a^{\,T}\vec a
=\boxed{14}.
\end{aligned}
\]

Finalmente, como los componentes son escalares, se pueden reordenar para reconocer el producto matricial:
\[
\begin{aligned}
\text{(e)}\qquad
S_{mn}a_ma_n
&=a_mS_{mn}a_n\\
&=(\vec a^{\,T})_mS_{mn}a_n\\
&=\vec a^{\,T}\mathbb S\vec a
=\boxed{59}.
\end{aligned}
\]

\subsection*{Problema 6}

Demuestre que si $\mathbb S$ es un tensor simétrico y $\mathbb A$ es un tensor antisimétrico, entonces
\[
S_{ij}A_{ij}=0.
\]

\textbf{Solución.} Sea $C=S_{ij}A_{ij}$. Al intercambiar los índices mudos $i$ y $j$ y usar $S_{ji}=S_{ij}$ y $A_{ji}=-A_{ij}$,
\[
C=S_{ji}A_{ji}=S_{ij}(-A_{ij})=-C.
\]
Por consiguiente, $2C=0$ y
\[
\boxed{S_{ij}A_{ij}=0}.
\]

\subsection*{Problema 8}

Demuestre que la traza y el determinante de un tensor son invariantes bajo transformaciones de coordenadas.

\textbf{Solución.} Para un cambio entre bases cartesianas ortonormales, sea $\mathbb R$ la matriz ortogonal de transformación. Entonces
\[
\mathbb T'=\mathbb R\mathbb T\mathbb R^T,
\qquad
R_{ik}R_{il}=\delta_{kl}.
\]
La traza se transforma como
\[
\mathrm{Tr}\,\mathbb T'
=T'_{ii}
=R_{ik}R_{il}T_{kl}
=\delta_{kl}T_{kl}
=\boxed{\mathrm{Tr}\,\mathbb T}.
\]
Además, como $\det\mathbb R=\pm1$,
\[
\det\mathbb T'
=\det\mathbb R\,\det\mathbb T\,\det\mathbb R^T
=(\det\mathbb R)^2\det\mathbb T
=\boxed{\det\mathbb T}.
\]

\subsection*{Problema 11}

El momento de cuadrupolo eléctrico $\mathbb Q$ de una distribución de carga $\rho(\vec r)$ se define como
\[
\mathbb Q
=\int \rho(\vec r\,')
\left[3\vec r\,'\vec r\,'-r'^2\mathbb I\right]dV'.
\]

\begin{enumerate}[(a)]
\item Demuestre que la traza de este tensor es cero.
\item Calcule el campo electrostático asociado con este cuadrupolo si el potencial se define como
\[
\Phi(\vec r)=\frac{1}{2r^5}\,\vec r\cdot\mathbb Q\cdot\vec r.
\]
\end{enumerate}

\textbf{Solución.} Para la traza, usando $\delta_{ii}=3$ y $r'_ir'_i=r'^2$,
\[
\begin{aligned}
Q_{ii}
&=\int\rho(\vec r\,')
\left(3r'_ir'_i-r'^2\delta_{ii}\right)dV'\\
&=\int\rho(\vec r\,')\left(3r'^2-3r'^2\right)dV'=\boxed{0}.
\end{aligned}
\]

Para calcular el campo, escribimos
\[
\Phi=\frac{1}{2}r^{-5}r_iQ_{ij}r_j.
\]
Como $\mathbb Q$ es simétrico,
\[
\frac{\partial}{\partial r_k}(r_iQ_{ij}r_j)=2Q_{kj}r_j,
\qquad
\frac{\partial r^{-5}}{\partial r_k}=-5r_k r^{-7}.
\]
Por tanto, $\vec E=-\vec\nabla\Phi$ tiene componentes
\[
E_k
=-\frac{Q_{kj}r_j}{r^5}
+\frac{5}{2r^7}(r_iQ_{ij}r_j)r_k.
\]
En forma vectorial,
\[
\boxed{
\vec E(\vec r)
=\frac{1}{2r^7}
\left[5(\vec r\cdot\mathbb Q\cdot\vec r)\vec r
-2r^2\mathbb Q\cdot\vec r\right]}.
\]
